**登录流程说明：**

1. 客户端发送 `POST /auth/login` 请求，携带 `{account, password}` 或 `{email, code}`
2. `JwtAuthFilter` 检测到无 JWT Token，直接放行
3. `SecurityConfig` 匹配到公开 URL 白名单，允许访问
4. `AuthController.login()` 调用 `AuthServiceImpl.login()`
5. `AuthServiceImpl` 根据是否含 `email+code` 字段选择密码登录或验证码登录
6. **密码登录**：通过 `UserMapper` 查询用户，使用 `BCryptPasswordEncoder.matches()` 校验密码
7. **验证码登录**：查询 `email_verification_code` 表，使用 BCrypt 匹配验证码，校验后标记 `used_at`
8. 校验用户状态是否为 `active`，禁用用户拒绝登录
9. 调用 `JwtUtil.createToken(userId, role, username)` 生成 JWT Token（载荷含 userId, role, username）
10. 更新 `last_login_at` 字段，返回 Token 和用户信息

**请求验证流程说明：**

1. 客户端在请求头携带 `Authorization: Bearer <token>`
2. `JwtAuthFilter.doFilterInternal()` 提取 Token，去除 `Bearer ` 前缀
3. 调用 `JwtUtil.validateToken()` 解析并校验 JWT（签名、有效期）
4. 解析出 `userId` 和 `role`，构建 `UsernamePasswordAuthenticationToken`
5. 设置到 `SecurityContextHolder`，标记为已认证
6. 链式调用放行，`SecurityConfig` 检查路径权限
7. 对于需要认证的接口（如 `/auth/profile`），`SecurityContext` 中已有用户身份
8. Controller 方法中通过 `SecurityUtil.getCurrentUserId()` 获取当前用户 ID

### 2.4 关键代码：JwtAuthFilter 核心逻辑

```java
@Component
@RequiredArgsConstructor
public class JwtAuthFilter extends OncePerRequestFilter {

    private final JwtUtil jwtUtil;

    @Override
    protected void doFilterInternal(HttpServletRequest request,
                                    HttpServletResponse response,
                                    FilterChain filterChain)
            throws ServletException, IOException {

        // 1. 从 Authorization 头提取 Token
        String token = extractToken(request);

        if (token != null && jwtUtil.validateToken(token)) {
            // 2. 解析用户信息
            Long userId = jwtUtil.getUserIdFromToken(token);
            String role = jwtUtil.getRoleFromToken(token);

            // 3. 构建 Spring Security 的认证对象
            UsernamePasswordAuthenticationToken authentication =
                new UsernamePasswordAuthenticationToken(
                    userId, null,
                    List.of(new SimpleGrantedAuthority("ROLE_" + role.toUpperCase()))
                );
            authentication.setDetails(userId);

            // 4. 设置到 SecurityContext
            SecurityContextHolder.getContext().setAuthentication(authentication);
        }

        // 5. 放行 —— 无 Token/Token 无效时视为未登录
        filterChain.doFilter(request, response);
    }
}
```

### 2.5 关键代码：JwtUtil

```java
@Component
public class JwtUtil {

    @Value("${jwt.secret}")
    private String secret;

    @Value("${jwt.expiration}")
    private long expiration;  // 604800000ms = 7天

    public String createToken(Long userId, String role, String username) {
        Date now = new Date();
        return Jwts.builder()
            .claim("userId", userId)
            .claim("role", role)
            .claim("username", username)
            .setIssuedAt(now)
            .setExpiration(new Date(now.getTime() + expiration))
            .signWith(Keys.hmacShaKeyFor(secret.getBytes(StandardCharsets.UTF_8)))
            .compact();
    }

    public boolean validateToken(String token) {
        try {
            Jwts.parser()
                .setSigningKey(Keys.hmacShaKeyFor(secret.getBytes(StandardCharsets.UTF_8)))
                .build()
                .parseClaimsJws(token);
            return true;
        } catch (Exception e) {
            return false;
        }
    }

    public Long getUserIdFromToken(String token) {
        return parseToken(token).get("userId", Long.class);
    }

    public String getRoleFromToken(String token) {
        return parseToken(token).get("role", String.class);
    }

    private Claims parseToken(String token) {
        return Jwts.parser()
            .setSigningKey(Keys.hmacShaKeyFor(secret.getBytes(StandardCharsets.UTF_8)))
            .build()
            .parseClaimsJws(token)
            .getBody();
    }
}
```

### 3.3 菜品搜索 SQL 实现（DishMapper.xml）

```xml
<select id="selectDishPage" resultMap="DishVOResultMap">
    SELECT d.*, s.name AS stall_name, s.canteen_id,
           c.name AS canteen_name
    FROM dish d
    JOIN stall s ON d.stall_id = s.id
    JOIN canteen c ON s.canteen_id = c.id
    <where>
        d.is_deleted = 0
        AND d.status = 'on'
        AND s.status = 'open'
        AND c.status = 'open'
        <if test="req.keyword != null and req.keyword != ''">
            AND (d.name LIKE CONCAT('%', #{req.keyword}, '%')
                 OR s.name LIKE CONCAT('%', #{req.keyword}, '%')
                 OR c.name LIKE CONCAT('%', #{req.keyword}, '%')
                 OR d.tags LIKE CONCAT('%', #{req.keyword}, '%'))
        </if>
        <if test="req.canteenId != null">
            AND s.canteen_id = #{req.canteenId}
        </if>
        <if test="req.stallId != null">
            AND d.stall_id = #{req.stallId}
        </if>
        <if test="req.tag != null and req.tag != ''">
            AND d.tags LIKE CONCAT('%', #{req.tag}, '%')
        </if>
        <if test="req.minPrice != null">
            AND d.price >= #{req.minPrice}
        </if>
        <if test="req.maxPrice != null">
            AND d.price &lt;= #{req.maxPrice}
        </if>
    </where>
    <choose>
        <when test="req.sortBy == 'rating'">
            ORDER BY d.avg_rating ${req.sortOrder}
        </when>
        <when test="req.sortBy == 'price'">
            ORDER BY d.price ${req.sortOrder}
        </when>
        <when test="req.sortBy == 'collects'">
            ORDER BY d.favorite_count ${req.sortOrder}
        </when>
        <when test="req.sortBy == 'created_at'">
            ORDER BY d.created_at ${req.sortOrder}
        </when>
        <otherwise>
            ORDER BY (d.favorite_count * 3 + d.view_count + d.rating_count * 5) DESC
        </otherwise>
    </choose>
</select>
```

```java
@Service
@RequiredArgsConstructor
public class ReviewServiceImpl implements ReviewService {

    private final ApplicationEventPublisher eventPublisher;

    @Override
    public void submitReview(Long userId, ReviewReq req) {
        // 1. 校验一人一菜一评
        Long count = reviewMapper.selectCount(
            new LambdaQueryWrapper<Review>()
                .eq(Review::getUserId, userId)
                .eq(Review::getDishId, req.getDishId())
                .eq(Review::getIsDeleted, 0));
        if (count > 0) {
            throw new BusinessException("您已经评价过该菜品");
        }

        // 2. 插入评价
        Review review = new Review();
        // ... 设置字段
        reviewMapper.insert(review);

        // 3. 发布事件（事务提交后触发）
        eventPublisher.publishEvent(
            new ReviewSubmittedEvent(this, req.getDishId(), req.getRating()));
    }
}
```

**评分重算（Dish 模块）：**
```java
@Service
public class DishServiceImpl implements DishService {

    @Override
    public void recalcAvgRating(Long dishId) {
        // 计算该菜品所有未隐藏评价的平均分
        List<Review> reviews = reviewMapper.selectList(
            new LambdaQueryWrapper<Review>()
                .eq(Review::getDishId, dishId)
                .eq(Review::getIsHidden, 0));
        
        if (reviews.isEmpty()) {
            dishMapper.updateById(
                new Dish().setId(dishId).setAvgRating(BigDecimal.ZERO).setRatingCount(0));
            return;
        }

        double avg = reviews.stream()
            .mapToInt(Review::getRating)
            .average()
            .orElse(0.0);

        Dish dish = new Dish();
        dish.setId(dishId);
        dish.setAvgRating(BigDecimal.valueOf(Math.round(avg * 10) / 10.0));
        dish.setRatingCount(reviews.size());
        dishMapper.updateById(dish);
    }
}
```

### 4.4 事件驱动机制设计原因

| 方案 | 缺点 | 本项目选择 |
|------|------|-----------|
| Review 直接调用 Dish 方法 | 模块紧耦合，违反"只依赖common"原则 | ❌ |
| Review 模块注入 DishService | Review 模块需依赖 Dish 模块 | ❌ |
| **Spring 事件（Event）** | 需要额外的事件类 | ✅ **选用** |
| 消息队列（RabbitMQ/Kafka） | 架构过重，demo 阶段不必要 | ❌ |

**选择 Spring 事件的原因：**
- 解耦：Review 模块不需要知道是谁在处理评分更新
- 轻量：无需引入外部中间件
- 事务安全：`@TransactionalEventListener(AFTER_COMMIT)` 确保事务提交后再触发

### 5.3 收藏切换核心代码

```java
@Service
@RequiredArgsConstructor
public class FavoriteServiceImpl implements FavoriteService {

    private final FavoriteMapper favoriteMapper;
    private final ApplicationEventPublisher eventPublisher;

    @Override
    public boolean toggle(Long userId, Long dishId) {
        // 1. 查询是否已收藏
        Favorite existing = favoriteMapper.selectOne(
            new LambdaQueryWrapper<Favorite>()
                .eq(Favorite::getUserId, userId)
                .eq(Favorite::getDishId, dishId));

        boolean isFavorited;
        if (existing != null) {
            // 2a. 已收藏 → 取消
            favoriteMapper.deleteById(existing.getId());
            isFavorited = false;
        } else {
            // 2b. 未收藏 → 新增
            Favorite favorite = new Favorite();
            favorite.setUserId(userId);
            favorite.setDishId(dishId);
            favoriteMapper.insert(favorite);
            isFavorited = true;
        }

        // 3. 发布事件（事务提交后触发收藏数同步）
        eventPublisher.publishEvent(
            new FavoriteChangedEvent(this, dishId, isFavorited));

        return isFavorited;
    }

    @Override
    public CollectResp batchCollect(Long userId, List<Long> dishIds) {
        int succeeded = 0, skipped = 0;
        for (Long dishId : dishIds) {
            if (isFavorited(userId, dishId)) {
                skipped++;
            } else {
                Favorite favorite = new Favorite();
                favorite.setUserId(userId);
                favorite.setDishId(dishId);
                favoriteMapper.insert(favorite);
                eventPublisher.publishEvent(
                    new FavoriteChangedEvent(this, dishId, true));
                succeeded++;
            }
        }
        return new CollectResp(succeeded, skipped);
    }
}
```

### 5.4 双事件监听对比

| 事件 | 监听器 | 所在模块 | 触发时机 | 回调方法 |
|------|--------|---------|---------|---------|
| `ReviewSubmittedEvent` | `RatingUpdateListener` | dish 模块 | 评价新增/修改/删除/隐藏 | `recalcAvgRating(dishId)` |
| `FavoriteChangedEvent` | `CollectCountListener` | dish 模块 | 收藏/取消收藏 | `syncCollectCount(dishId)` |

**共同特点：**
- 都使用 `@TransactionalEventListener(phase = AFTER_COMMIT)` 确保事务提交后执行
- 都在 dish 模块中监听，由 dish 模块的 Service 执行实际数据更新
- 都是同步事件（demo 阶段无需消息队列）

```java
@Service
public class UploadServiceImpl implements UploadService {

    @Value("${upload.path}")
    private String uploadPath;           // 默认 ./uploads/images

    @Value("${upload.url-prefix}")
    private String urlPrefix;            // 默认 /images

    private final ImageUrlUtil imageUrlUtil;

    private static final Set<String> ALLOWED_EXTENSIONS =
        Set.of("jpg", "jpeg", "png", "webp");

    @Override
    public Map<String, String> uploadImage(MultipartFile file) {
        // 1. 校验文件非空
        if (file == null || file.isEmpty()) {
            throw new BusinessException("上传文件不能为空");
        }

        // 2. 校验扩展名
        String originalFilename = file.getOriginalFilename();
        String extension = originalFilename.substring(
            originalFilename.lastIndexOf(".") + 1).toLowerCase();
        if (!ALLOWED_EXTENSIONS.contains(extension)) {
            throw new BusinessException("仅支持 jpg/jpeg/png/webp 格式");
        }

        // 3. 生成存储路径：/yyyy/MM/uuid.ext
        String dateDir = LocalDate.now().format(DateTimeFormatter.ofPattern("yyyy/MM"));
        String uuid = UUID.randomUUID().toString().replace("-", "");
        String fileName = uuid + "." + extension;

        // 4. 创建目录并保存文件
        File targetDir = new File(uploadPath, dateDir);
        if (!targetDir.exists()) {
            targetDir.mkdirs();
        }
        File targetFile = new File(targetDir, fileName);
        file.transferTo(targetFile);

        // 5. 构建 URL
        String relativeUrl = urlPrefix + "/" + dateDir + "/" + fileName;
        String absoluteUrl = imageUrlUtil.toAbsoluteUrl(relativeUrl);

        // 6. 返回结果
        return Map.of(
            "url", absoluteUrl,
            "relativeUrl", relativeUrl
        );
    }
}
```

```java
@RestControllerAdvice
public class GlobalExceptionHandler {

    // 业务异常
    @ExceptionHandler(BusinessException.class)
    public Result<Void> handleBusinessException(BusinessException e) {
        return Result.badRequest(e.getMessage());
    }

    // 未登录
    @ExceptionHandler(UnauthorizedException.class)
    public Result<Void> handleUnauthorizedException(UnauthorizedException e) {
        return Result.unauthorized(e.getMessage());
    }

    // 参数校验失败（@Valid）
    @ExceptionHandler(MethodArgumentNotValidException.class)
    public Result<Void> handleValidationException(MethodArgumentNotValidException e) {
        String message = e.getBindingResult()
            .getFieldErrors().stream()
            .map(f -> f.getDefaultMessage())
            .findFirst()
            .orElse("参数校验失败");
        return Result.badRequest(message);
    }

    // 兜底异常（500）
    @ExceptionHandler(Exception.class)
    public Result<Void> handleException(Exception e) {
        log.error("服务器内部错误", e);
        return Result.error("服务器繁忙，请稍后再试");
    }
}
```
